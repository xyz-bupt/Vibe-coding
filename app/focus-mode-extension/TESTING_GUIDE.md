## 自动化测试脚本

创建自动化测试文件: `/Users/abc/Vibe-coding/app/focus-mode-extension/automated-tests.js`

```javascript
// 自动化测试脚本
// 使用方式: 在浏览器控制台运行

(async function runAutomatedTests() {
  const testResults = [];

  function testCase(name, fn) {
    return {
      name,
      run: async () => {
        try {
          await fn();
          return { passed: true };
        } catch (error) {
          return { passed: false, error: error.message };
        }
      }
    };
  }

  const tests = [
    testCase('内容选择器测试', async () => {
      const selectors = window.FocusModeFixes.IMPROVED_CONTENT_SELECTORS;
      if (!Array.isArray(selectors) || selectors.length === 0) {
        throw new Error('选择器数组为空');
      }
      console.log(`✓ 找到 ${selectors.length} 个内容选择器`);
    }),

    testCase('智能提取算法测试', async () => {
      const result = window.FocusModeFixes.improvedIntelligentlyExtractContent();
      if (!result) {
        throw new Error('未能提取内容');
      }
      console.log('✓ 智能提取成功');
    }),

    testCase('内容清理测试', async () => {
      const testDiv = document.createElement('div');
      testDiv.innerHTML = '<p>测试内容</p><script>alert("xss")</script>';
      window.FocusModeFixes.improvedCleanContent(testDiv);
      if (testDiv.querySelector('script')) {
        throw new Error('脚本未被移除');
      }
      console.log('✓ 内容清理成功');
    })
  ];

  // 运行所有测试
  for (const test of tests) {
    const result = await test.run();
    testResults.push({
      name: test.name,
      ...result
    });
  }

  // 输出结果
  console.table(testResults);

  const passed = testResults.filter(r => r.passed).length;
  const total = testResults.length;

  console.log(`\n测试完成: ${passed}/${total} 通过`);

  return testResults;
})();
```

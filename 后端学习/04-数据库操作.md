## 什么是 ORM？

ORM（Object-Relational Mapping，对象关系映射）是一种技术，它将数据库中的表映射为 Python 类，将行映射为对象。

### 为什么使用 ORM？

**不使用 ORM（原生 SQL）**：
```python
import sqlite3

# 创建连接
conn = sqlite3.connect('database.db')
cursor = conn.cursor()

# 执行 SQL
cursor.execute('INSERT INTO users (name, email) VALUES (?, ?)', ('张三', 'zhangsan@example.com'))
conn.commit()

# 查询数据
cursor.execute('SELECT * FROM users WHERE name = ?', ('张三',))
user = cursor.fetchone()
```

**使用 ORM**：
```python
# 创建用户
user = User(name='张三', email='zhangsan@example.com')
db.session.add(user)
db.session.commit()

# 查询用户
user = User.query.filter_by(name='张三').first()
```

**ORM 的优势**：
- 面向对象，更 Pythonic
- 自动处理 SQL 注入
- 数据库无关性（易于切换数据库）
- 代码更易维护
- 自动处理事务

## SQLAlchemy 简介

SQLAlchemy 是 Python 最流行的 ORM 框架，Flask-SQLAlchemy 是它的 Flask 集成。

### 安装

```bash
pip install flask-sqlalchemy
```

## 模型定义

### 基本模型

```python
from flask_sqlalchemy import SQLAlchemy
from datetime import datetime

db = SQLAlchemy()

class User(db.Model):
    """用户模型"""
    __tablename__ = 'users'  # 表名（可选，默认使用类名的小写形式）

    # 定义列
    id = db.Column(db.Integer, primary_key=True)  # 主键
    name = db.Column(db.String(80), nullable=False)  # 字符串，不能为空
    email = db.Column(db.String(120), unique=True, nullable=False)  # 唯一，不能为空
    created_at = db.Column(db.DateTime, default=datetime.now)  # 日期时间，默认值
    active = db.Column(db.Boolean, default=True)  # 布尔值

    def __repr__(self):
        """对象的字符串表示"""
        return f'<User {self.name}>'
```

### 常用列类型

```python
# 整数
age = db.Column(db.Integer)

# 字符串
name = db.Column(db.String(80))  # 最大长度 80
text = db.Column(db.Text)  # 长文本

# 浮点数
price = db.Column(db.Float)
salary = db.Column(db.Decimal(10, 2))  # 总共10位，小数点后2位

# 布尔值
is_active = db.Column(db.Boolean, default=True)

# 日期时间
created_at = db.Column(db.DateTime)
birth_date = db.Column(db.Date)
meeting_time = db.Column(db.Time)

# 二进制数据
avatar = db.Column(db.LargeBinary)

# JSON（PostgreSQL 或 SQLite）
metadata = db.Column(db.JSON)
```

### 列选项

```python
# 主键
id = db.Column(db.Integer, primary_key=True)

# 不能为空
name = db.Column(db.String(80), nullable=False)

# 唯一
email = db.Column(db.String(120), unique=True)

# 默认值

# 索引

# 自动递增

```

### 完整示例：博客模型

```python
from flask_sqlalchemy import SQLAlchemy
from datetime import datetime

db = SQLAlchemy()

class User(db.Model):
    """用户模型"""
    __tablename__ = 'users'

    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False, index=True)
    email = db.Column(db.String(120), unique=True, nullable=False)
    password_hash = db.Column(db.String(128))
    avatar = db.Column(db.String(200), default='default.jpg')
    bio = db.Column(db.Text)
    is_active = db.Column(db.Boolean, default=True)
    created_at = db.Column(db.DateTime, default=datetime.now)
    last_login = db.Column(db.DateTime)

    # 关系：一个用户有多篇文章
    posts = db.relationship('Post', backref='author', lazy='dynamic')

    def __repr__(self):
        return f'<User {self.username}>'

class Post(db.Model):
    """文章模型"""
    __tablename__ = 'posts'

    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(200), nullable=False)
    content = db.Column(db.Text, nullable=False)
    summary = db.Column(db.String(300))
    published = db.Column(db.Boolean, default=False)
    views = db.Column(db.Integer, default=0)
    created_at = db.Column(db.DateTime, default=datetime.now)
    updated_at = db.Column(db.DateTime, default=datetime.now, onupdate=datetime.now)

    # 外键：每篇文章属于一个用户
    user_id = db.Column(db.Integer, db.ForeignKey('users.id'), nullable=False)

    # 关系：一篇文章有多个标签
    tags = db.relationship('Tag', secondary='post_tags', backref=db.backref('posts', lazy='dynamic'))

    def __repr__(self):
        return f'<Post {self.title}>'

class Tag(db.Model):
    """标签模型"""
    __tablename__ = 'tags'

    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(50), unique=True, nullable=False)
    created_at = db.Column(db.DateTime, default=datetime.now)

    def __repr__(self):
        return f'<Tag {self.name}>'

# 多对多关系表（文章和标签）
post_tags = db.Table('post_tags',
    db.Column('post_id', db.Integer, db.ForeignKey('posts.id'), primary_key=True),
    db.Column('tag_id', db.Integer, db.ForeignKey('tags.id'), primary_key=True)
)
```

## CRUD 操作

### Create - 创建记录

```python
# 方法 1：创建对象并添加
user = User(username='zhangsan', email='zhangsan@example.com')
db.session.add(user)
db.session.commit()

# 方法 2：使用构造函数
user = User()
user.username = 'lisi'
user.email = 'lisi@example.com'
db.session.add(user)
db.session.commit()

# 方法 3：批量添加
users = [
    User(username='user1', email='user1@example.com'),
    User(username='user2', email='user2@example.com'),
    User(username='user3', email='user3@example.com'),
]
db.session.add_all(users)
db.session.commit()
```

### Read - 查询记录

```python
# 查询所有用户
all_users = User.query.all()

# 查询第一个用户
first_user = User.query.first()

# 根据 ID 查询
user = User.query.get(1)  # SQLAlchemy 1.4+
user = User.query.get_or_404(1)  # 如果不存在返回 404

# 使用 filter_by（简单条件）
user = User.query.filter_by(username='zhangsan').first()
users = User.query.filter_by(is_active=True).all()

# 使用 filter（复杂条件）
user = User.query.filter(User.username == 'zhangsan').first()
users = User.query.filter(User.email.like('%@example.com')).all()

# 多条件查询
from sqlalchemy import or_, and_

users = User.query.filter(
    and_(User.is_active == True, User.created_at >= datetime(2024, 1, 1))
).all()

users = User.query.filter(
    or_(User.username == 'admin', User.email == 'admin@example.com')
).first()

# 排序
users = User.query.order_by(User.created_at.desc()).all()  # 降序
users = User.query.order_by(User.created_at.asc()).all()  # 升序

# 限制数量
users = User.query.limit(10).all()  # 前 10 条
users = User.query.offset(10).limit(10).all()  # 跳过前 10 条，取 10 条

# 统计
count = User.query.count()
count = User.query.filter_by(is_active=True).count()

# 聚合查询
from sqlalchemy import func

max_views = db.session.query(func.max(Post.views)).first()
avg_views = db.session.query(func.avg(Post.views)).first()
```

### Update - 更新记录

```python
# 方法 1：修改对象属性
user = User.query.get(1)
user.email = 'newemail@example.com'
db.session.commit()

# 方法 2：批量更新
User.query.filter_by(is_active=False).update({'is_active': True})
db.session.commit()

# 方法 3：使用 update 语句
User.query.filter(User.username.like('test%')).update(
    {User.is_active: False},
    synchronize_session=False
)
db.session.commit()
```

### Delete - 删除记录

```python
# 删除单个对象
user = User.query.get(1)
db.session.delete(user)
db.session.commit()

# 批量删除
User.query.filter_by(is_active=False).delete()
db.session.commit()

# 删除所有（谨慎！）
User.query.delete()
db.session.commit()
```

## 关系映射

### 一对多关系

一个用户有多篇文章：

```python
class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), nullable=False)

    # 关系定义
    posts = db.relationship('Post', backref='author', lazy=True)

class Post(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(200), nullable=False)

    # 外键
    user_id = db.Column(db.Integer, db.ForeignKey('users.id'), nullable=False)

# 使用
user = User.query.get(1)
# 获取用户的所有文章
user_posts = user.posts

# 创建文章并关联用户
post = Post(title='新文章', author=user)  # 使用 backref
db.session.add(post)
db.session.commit()
```

**relationship 参数说明**：
- `backref` - 在反向模型中添加关系属性
- `lazy` - 加载方式
  - `select` - 延迟加载（默认）
  - `joined` - 联合加载
  - `subquery` - 子查询加载
  - `dynamic` - 返回查询对象（可继续过滤）
  - `True` - 等同于 `select`
  - `False` - 立即加载

### 多对多关系

文章和标签：

```python
# 关联表
post_tags = db.Table('post_tags',
    db.Column('post_id', db.Integer, db.ForeignKey('posts.id'), primary_key=True),
    db.Column('tag_id', db.Integer, db.ForeignKey('tags.id'), primary_key=True)
)

class Post(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(200), nullable=False)

    # 多对多关系
    tags = db.relationship('Tag', secondary=post_tags, backref='posts')

class Tag(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(50), nullable=False)

# 使用
post = Post.query.get(1)
tag1 = Tag.query.get(1)
tag2 = Tag.query.get(2)

# 添加标签
post.tags.append(tag1)
post.tags.append(tag2)
db.session.commit()

# 移除标签
post.tags.remove(tag1)
db.session.commit()

# 获取文章的所有标签
post_tags = post.tags

# 获取标签的所有文章
tag_posts = tag1.posts
```

用户和详细资料：

```python
class User(db.Model):

    # 一对一关系

```

## 数据库迁移

Flask-Migrate 是基于 Alembic 的数据库迁移工具，用于管理数据库结构的变化。

### 安装

```bash
pip install flask-migrate
```

### 配置

```python
from flask import Flask
from flask_sqlalchemy import SQLAlchemy
from flask_migrate import Migrate

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///blog.db'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False

db = SQLAlchemy(app)
migrate = Migrate(app, db)
```

### 使用迁移

```bash
# 初始化迁移仓库（只需执行一次）
flask db init

# 生成迁移脚本
flask db migrate -m "添加用户表"

# 应用迁移
flask db upgrade

# 回滚迁移
flask db downgrade

# 查看迁移历史
flask db history

# 查看当前版本
flask db current
```

### 迁移工作流程

1. **修改模型**
```python
class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), nullable=False)
    email = db.Column(db.String(120), nullable=False)
    # 新增字段
    phone = db.Column(db.String(20))
```

2. **生成迁移**
```bash
flask db migrate -m "添加用户手机号字段"
```

3. **检查生成的迁移文件**
生成的文件在 `migrations/versions/` 目录下，检查是否正确。

4. **应用迁移**
```bash
flask db upgrade
```

## 完整示例：用户管理系统

```python
from flask import Flask, render_template, request, redirect, url_for, flash
from flask_sqlalchemy import SQLAlchemy
from datetime import datetime

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///users.db'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False
app.secret_key = 'your-secret-key'

db = SQLAlchemy(app)

# 模型定义
class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False)
    email = db.Column(db.String(120), unique=True, nullable=False)
    age = db.Column(db.Integer)
    city = db.Column(db.String(50))
    created_at = db.Column(db.DateTime, default=datetime.now)

    def __repr__(self):
        return f'<User {self.username}>'

# 创建数据库
with app.app_context():
    db.create_all()

# 路由
@app.route('/')
def index():
    # 分页查询
    page = request.args.get('page', 1, type=int)
    per_page = 10
    users = User.query.order_by(User.created_at.desc()).paginate(
        page=page, per_page=per_page, error_out=False
    )
    return render_template('index.html', users=users)

@app.route('/user/create', methods=['GET', 'POST'])
def create_user():
    if request.method == 'POST':
        username = request.form.get('username')
        email = request.form.get('email')
        age = request.form.get('age', type=int)
        city = request.form.get('city')

        # 检查用户名是否存在
        if User.query.filter_by(username=username).first():
            flash('用户名已存在', 'danger')
            return redirect(url_for('create_user'))

        # 创建用户
        user = User(username=username, email=email, age=age, city=city)
        db.session.add(user)
        db.session.commit()

        flash('用户创建成功', 'success')
        return redirect(url_for('index'))

    return render_template('create.html')

@app.route('/user/<int:user_id>')
def user_detail(user_id):
    user = User.query.get_or_404(user_id)
    return render_template('detail.html', user=user)

@app.route('/user/<int:user_id>/edit', methods=['GET', 'POST'])
def edit_user(user_id):
    user = User.query.get_or_404(user_id)

    if request.method == 'POST':
        user.username = request.form.get('username')
        user.email = request.form.get('email')
        user.age = request.form.get('age', type=int)
        user.city = request.form.get('city')

        db.session.commit()
        flash('用户更新成功', 'success')
        return redirect(url_for('user_detail', user_id=user.id))

    return render_template('edit.html', user=user)

@app.route('/user/<int:user_id>/delete', methods=['POST'])
def delete_user(user_id):
    user = User.query.get_or_404(user_id)
    db.session.delete(user)
    db.session.commit()
    flash('用户删除成功', 'success')
    return redirect(url_for('index'))

@app.route('/search')
def search():
    query = request.args.get('q', '')
    users = User.query.filter(
        db.or_(
            User.username.like(f'%{query}%'),
            User.email.like(f'%{query}%'),
            User.city.like(f'%{query}%')
        )
    ).all()
    return render_template('search.html', users=users, query=query)

if __name__ == '__main__':
    app.run(debug=True)
```

## 常见问题

### Q1: 如何处理数据库迁移中的错误？

```bash
# 如果迁移失败，检查当前状态
flask db current

# 回滚到上一个版本
flask db downgrade

# 修复后重新生成迁移
flask db migrate -m "修复迁移"
```

### Q2: 如何优化查询性能？

```python
# 使用 joinedload 避免N+1查询
from sqlalchemy.orm import joinedload

# 不好的做法（N+1查询）
posts = Post.query.all()
for post in posts:
    print(post.author.username)  # 每次都查询数据库

# 好的做法（联合查询）
posts = Post.query.options(joinedload(Post.author)).all()
for post in posts:
    print(post.author.username)  # 只查询一次
```

### Q3: 如何执行原生 SQL？

```python
# 执行原生 SQL
result = db.session.execute('SELECT * FROM users WHERE age > :age', {'age': 18})
users = result.fetchall()

# 执行更新
db.session.execute('UPDATE users SET city = :city WHERE id = :id',
                  {'city': '北京', 'id': 1})
db.session.commit()
```

## 最佳实践

### 1. 使用索引优化查询

```python
class User(db.Model):
    username = db.Column(db.String(80), unique=True, nullable=False, index=True)
    email = db.Column(db.String(120), unique=True, nullable=False, index=True)
    created_at = db.Column(db.DateTime, default=datetime.now, index=True)
```

### 2. 使用查询对象而非执行查询

```python
# 好的做法
query = User.query.filter_by(is_active=True)
if search_term:
    query = query.filter(User.username.like(f'%{search_term}%'))
users = query.all()

# 不好的做法（过早执行）
users = User.query.filter_by(is_active=True).all()
```

### 3. 使用事务

```python
try:
    # 多个数据库操作
    user = User(username='test', email='test@example.com')
    db.session.add(user)

    profile = UserProfile(user=user, bio='简介')
    db.session.add(profile)

    # 提交事务
    db.session.commit()
except Exception as e:
    # 回滚事务
    db.session.rollback()
    raise
```

## 蓝图（Blueprints）

蓝图是 Flask 组织应用的方式，可以将应用分割为多个模块。

### 为什么使用蓝图？

- 组织代码：将相关功能分组
- 可复用性：蓝图中注册的视图、模板等可在多个应用间复用
- 大型项目：便于团队协作开发

### 创建蓝图

```python
from flask import Blueprint, render_template, jsonify

# 创建蓝图
auth_bp = Blueprint('auth', __name__, url_prefix='/auth')

# 定义路由
@auth_bp.route('/login')
def login():
    return render_template('auth/login.html')

@auth_bp.route('/register')
def register():
    return render_template('auth/register.html')

# API 蓝图
api_bp = Blueprint('api', __name__, url_prefix='/api')

@api_bp.route('/users')
def get_users():
    users = [{'id': 1, 'name': '张三'}, {'id': 2, 'name': '李四'}]
    return jsonify(users)
```

### 注册蓝图

```python
from flask import Flask

app = Flask(__name__)

# 注册蓝图
app.register_blueprint(auth_bp)
app.register_blueprint(api_bp)

if __name__ == '__main__':
    app.run(debug=True)
```

## 请求钩子（中间件）

请求钩子允许你在请求处理的不同阶段执行代码。

### 钩子类型

```python
from flask import Flask, request, g

app = Flask(__name__)

# before_first_request：第一个请求之前执行（已废弃，使用 before_request 替代）
@app.before_request
def before_first():
    print("每个请求前执行")

# before_request：每个请求前执行
@app.before_request
def before_request():
    print(f"请求路径：{request.path}")
    # 可以在这里做认证、日志等

    # 设置 g 对象（请求级别的全局变量）
    g.user = None

# after_request：每个请求后执行（如果没有异常）
@app.after_request
def after_request(response):
    print("请求后执行")
    response.headers['X-Custom-Header'] = 'Value'
    return response

# teardown_request：每个请求后执行（即使有异常）
@app.teardown_request
def teardown_request(exception):
    print("清理资源")
    if exception:
        print(f"发生异常：{exception}")

# teardown_appcontext：应用上下文 teardown
@app.teardown_appcontext
def teardown_db(exception):
    # 清理数据库连接等
    pass

# errorhandler：错误处理
@app.errorhandler(404)
def not_found(error):
    return "页面未找到", 404

@app.errorhandler(Exception)
def handle_error(error):
    print(f"错误：{error}")
    return "服务器错误", 500
```

### 实际应用示例

```python
# 认证中间件
@app.before_request
def check_auth():
    # 不需要认证的路由
    public_paths = ['/login', '/register', '/static']
    if request.path in public_paths:
        return

    # 检查用户是否登录
    if 'user_id' not in session:
        return redirect(url_for('login'))

# 请求日志中间件
@app.before_request
def log_request():
    g.start_time = time.time()

@app.after_request
def log_response(response):
    if hasattr(g, 'start_time'):
        duration = time.time() - g.start_time
        print(f"请求 {request.path} 耗时 {duration:.3f}秒")
    return response
```

## 文件上传

### 基本上传

```python
from flask import Flask, request, render_template, redirect, url_for
from werkzeug.utils import secure_filename
import os

app = Flask(__name__)
app.config['UPLOAD_FOLDER'] = '/path/to/uploads'
app.config['MAX_CONTENT_LENGTH'] = 16 * 1024 * 1024  # 16MB
ALLOWED_EXTENSIONS = {'png', 'jpg', 'jpeg', 'gif', 'pdf'}

def allowed_file(filename):
    return '.' in filename and \
           filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

@app.route('/upload', methods=['GET', 'POST'])
def upload_file():
    if request.method == 'POST':
        # 检查是否有文件
        if 'file' not in request.files:
            return '没有文件', 400

        file = request.files['file']

        # 检查文件名
        if file.filename == '':
            return '未选择文件', 400

        # 检查文件类型
        if file and allowed_file(file.filename):
            filename = secure_filename(file.filename)
            filepath = os.path.join(app.config['UPLOAD_FOLDER'], filename)
            file.save(filepath)
            return f'文件已上传：{filename}'

        return '不支持的文件类型', 400

    return render_template('upload.html')
```

### 多文件上传

```python
@app.route('/uploads', methods=['POST'])
def upload_files():
    files = request.files.getlist('files')
    uploaded_files = []

    for file in files:
        if file and allowed_file(file.filename):
            filename = secure_filename(file.filename)
            file.save(os.path.join(app.config['UPLOAD_FOLDER'], filename))
            uploaded_files.append(filename)

    return f'已上传 {len(uploaded_files)} 个文件'
```

```python
from flask_mail import Mail, Message

app.config['MAIL_SERVER'] = 'smtp.gmail.com'
app.config['MAIL_PORT'] = 587
app.config['MAIL_USE_TLS'] = True
app.config['MAIL_USERNAME'] = 'your-email@gmail.com'
app.config['MAIL_PASSWORD'] = 'your-password'

mail = Mail(app)

# 发送邮件
def send_email(to, subject, template):
    msg = Message(
        subject,
        sender='your-email@gmail.com',
        recipients=[to]
    )
    msg.html = render_template(template)
    mail.send(msg)
```

## Celery 异步任务

### 安装

```bash
pip install celery redis
```

### 配置

```python
from celery import Celery

app.config['CELERY_BROKER_URL'] = 'redis://localhost:6379/0'
app.config['CELERY_RESULT_BACKEND'] = 'redis://localhost:6379/0'

# 创建 Celery 实例
celery = Celery(
    app.name,
    broker=app.config['CELERY_BROKER_URL']
)
celery.conf.update(app.config)
```

### 定义任务

```python
@celery.task
def send_async_email(to, subject, body):
    """异步发送邮件"""
    # 发送邮件的代码
    time.sleep(5)  # 模拟耗时操作
    print(f"邮件已发送至 {to}")
    return f"已发送邮件至 {to}"

@celery.task
def process_video(video_path):
    """异步处理视频"""
    # 处理视频的代码
    return f"视频处理完成：{video_path}"
```

### 调用任务

```python
@app.route('/send-email')
def send_email():
    # 异步发送
    task = send_async_email.delay('user@example.com', '欢迎', '欢迎注册')
    return f'邮件发送中，任务ID：{task.id}'

@app.route('/check-task/<task_id>')
def check_task(task_id):
    from celery.result import AsyncResult
    task = AsyncResult(task_id)

    if task.state == 'PENDING':
        response = {'state': task.state, 'status': '等待中...'}
    elif task.state == 'SUCCESS':
        response = {'state': task.state, 'result': task.result}
    else:
        response = {'state': task.state}

    return jsonify(response)
```

## 性能优化

### 1. 数据库查询优化

```python
# 使用 joinedload 避免 N+1 查询
from sqlalchemy.orm import joinedload

# 不好（N+1 查询）
posts = Post.query.all()
for post in posts:
    print(post.author.name)  # 每次都查询数据库

# 好（联合查询）
posts = Post.query.options(joinedload(Post.author)).all()
for post in posts:
    print(post.author.name)  # 只查询一次
```
